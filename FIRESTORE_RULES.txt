rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null && request.auth.token.email == 'camilosimanca77@gmail.com';
    }
    // Cuenta que usa GitHub Actions (notificaciones e historial privado).
    function isBot() {
      return request.auth != null && request.auth.token.email == 'bot@camilorey-bot.app';
    }
    // Premium: membresía en members/{correo en minúsculas} con "until" vigente.
    function myEmail() {
      return request.auth.token.email.lower();
    }
    function isPremium() {
      return request.auth != null && request.auth.token.email != null
        && exists(/databases/$(database)/documents/members/$(myEmail()))
        && get(/databases/$(database)/documents/members/$(myEmail())).data.until > request.time;
    }

    // Contadores de visitas: cualquiera puede sumar la suya (de a 1), pero
    // solo el admin puede leerlos.
    match /stats/{doc} {
      allow read: if isAdmin();
      allow create: if request.resource.data.keys().hasOnly(['count'])
                    && request.resource.data.count == 1;
      allow update: if request.resource.data.keys().hasOnly(['count'])
                    && request.resource.data.count == resource.data.count + 1;
    }

    // Usuarios: cada uno lee y escribe el suyo. El admin los puede leer
    // (panel), y el bot los lee para notificar y solo puede tocar "push"
    // (borrar suscripciones vencidas).
    match /users/{uid} {
      allow read: if (request.auth != null && request.auth.uid == uid) || isAdmin() || isBot();
      allow create, delete: if request.auth != null && request.auth.uid == uid;
      allow update: if (request.auth != null && request.auth.uid == uid)
                    || (isBot() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['push']));
    }

    // Bankroll del sistema e historial de apuestas: solo lo lee el admin y
    // solo lo escribe el bot.
    match /admin/{doc} {
      allow read: if isAdmin();
      allow create, update: if isBot();
    }

    // Picks del tipster: solo miembros (el sitio es privado); los publica el
    // admin y el bot los lee para calcular su efectividad.
    match /tipster/{doc} {
      allow read: if isAdmin() || isPremium() || isBot();
      allow write: if isAdmin();
    }

    // Llave para abrir los datos cifrados del sitio: solo la leen cuentas
    // con acceso vigente y el admin; la cambia el bot (cada día y cuando a
    // alguien se le vence o se le quita el acceso).
    match /vault/{doc} {
      allow read: if isAdmin() || isPremium() || isBot();
      allow create, update: if isBot();
    }

    // Membresías Premium: las activa el admin; cada usuario ve la suya;
    // el bot las lee para saber a quién mandar notificaciones.
    match /members/{email} {
      allow read: if isAdmin() || isBot() || (request.auth != null && myEmail() == email);
      allow write: if isAdmin();
    }

    // Contenido Premium (picks de alta confianza y Análisis IA): solo
    // cuentas con Premium activo (o admin); lo escribe el bot.
    match /premium/{doc} {
      allow read: if isAdmin() || isPremium();
      allow create, update: if isBot();
    }

    // Soporte en vivo: cada usuario solo su conversación (id = su uid);
    // el admin las ve y responde todas; el bot las lee para avisarle al
    // admin por notificación cuando alguien pide hablar con una persona.
    match /chats/{uid} {
      allow read: if (request.auth != null && request.auth.uid == uid) || isAdmin() || isBot();
      allow create, update: if (request.auth != null && request.auth.uid == uid) || isAdmin();
      allow delete: if isAdmin();
    }
  }
}
