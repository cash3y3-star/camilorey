rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null && request.auth.token.email == 'camilosimanca77@gmail.com';
    }
    // Cuenta que usa GitHub Actions (notificaciones e historial privado).
    function isBot() {
      return request.auth != null && request.auth.token.email == 'bot@camilorey-bot.app';
    }

    // Contadores públicos: cualquiera los lee, y solo se pueden sumar de a 1.
    match /stats/{doc} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['count'])
                    && request.resource.data.count == 1;
      allow update: if request.resource.data.keys().hasOnly(['count'])
                    && request.resource.data.count == resource.data.count + 1;
    }

    // Usuarios: cada uno lee y escribe el suyo. El admin los puede leer
    // (panel), y el bot los lee para notificar y solo puede tocar "push"
    // (borrar suscripciones vencidas).
    match /users/{uid} {
      allow read: if (request.auth != null && request.auth.uid == uid) || isAdmin() || isBot();
      allow create, delete: if request.auth != null && request.auth.uid == uid;
      allow update: if (request.auth != null && request.auth.uid == uid)
                    || (isBot() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['push']));
    }

    // Bankroll del sistema e historial de apuestas: solo lo lee el admin y
    // solo lo escribe el bot.
    match /admin/{doc} {
      allow read: if isAdmin();
      allow create, update: if isBot();
    }

    // Picks del tipster: los ve todo el mundo, solo los publica el admin.
    match /tipster/{doc} {
      allow read: if true;
      allow write: if isAdmin();
    }
  }
}
