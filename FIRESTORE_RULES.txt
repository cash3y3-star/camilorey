rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Contadores públicos: cualquiera los lee, y solo se pueden sumar de a 1.
    match /stats/{doc} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['count'])
                    && request.resource.data.count == 1;
      allow update: if request.resource.data.keys().hasOnly(['count'])
                    && request.resource.data.count == resource.data.count + 1;
    }

    // Cada usuario solo puede leer y escribir su propio documento.
    match /users/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

    // Historial de apuestas: solo lo lee la cuenta admin; solo lo escribe
    // GitHub Actions con la clave del bot (secret FIRESTORE_BOT_KEY).
    // CLAVE_DEL_BOT se reemplaza por la clave real al pegar en Firebase
    // (nunca se sube al repo, que es público).
    match /admin/{doc} {
      allow read: if request.auth != null
                  && request.auth.token.email == 'camilosimanca77@gmail.com';
      allow create, update: if request.resource.data.botKey == 'CLAVE_DEL_BOT';
    }
  }
}
